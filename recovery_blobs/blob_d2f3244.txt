        """Update hyperparameter widgets when model type changes."""

        self._tab7_create_hyperparam_widgets(model_type)

    def _tab7_update_wavelength_count(self, event=None):
        """Update wavelength count label in real-time."""
        try:
            wl_text = self.tab7_wl_spec.get('1.0', 'end-1c').strip()
            if not wl_text:
                self.tab7_wl_count_label.config(text="Count: 0 wavelengths", foreground='#999999')
                return

            wavelengths = self._parse_wavelength_spec(wl_text)
            count = len(wavelengths)

            if count > 0:
                self.tab7_wl_count_label.config(
                    text=f"Count: {count} wavelengths",
                    foreground='#2E7D32')
            else:
                self.tab7_wl_count_label.config(
                    text="Count: 0 wavelengths (check format)",
                    foreground='#D32F2F')
        except Exception as e:
            self.tab7_wl_count_label.config(
                text="Count: Invalid format",
                foreground='#D32F2F')

    def _tab7_apply_wl_preset(self, preset):
        """Apply wavelength presets."""
        if not hasattr(self, 'wavelengths') or self.wavelengths is None:
            messagebox.showwarning("No Data",
                "Please load spectral data first in the Import tab.")
            return

        wl = self.wavelengths

        if preset == 'all':
            wl_text = ', '.join(map(str, wl))
        elif preset == 'nir':
            nir_wl = [w for w in wl if w > 780]
            if not nir_wl:
                messagebox.showinfo("NIR Preset", "No NIR wavelengths (>780nm) found in data.")
                return
            wl_text = ', '.join(map(str, nir_wl))
        elif preset == 'visible':
            vis_wl = [w for w in wl if 400 <= w <= 780]
            if not vis_wl:
                messagebox.showinfo("Visible Preset", "No visible wavelengths (400-780nm) found in data.")
                return
            wl_text = ', '.join(map(str, vis_wl))
        else:
            return

        self.tab7_wl_spec.delete('1.0', 'end')
        self.tab7_wl_spec.insert('1.0', wl_text)
        self._tab7_update_wavelength_count()

    def _tab7_custom_range_dialog(self):
        """Open dialog for custom wavelength range."""
        if not hasattr(self, 'wavelengths') or self.wavelengths is None:
            messagebox.showwarning("No Data",
                "Please load spectral data first in the Import tab.")
            return

        dialog = tk.Toplevel(self.root)
        dialog.title("Custom Wavelength Range")
        dialog.geometry("400x200")

        ttk.Label(dialog, text="Enter wavelength range:",
                  style='Subheading.TLabel').pack(pady=10)

        range_frame = ttk.Frame(dialog)
        range_frame.pack(pady=10)

        ttk.Label(range_frame, text="From:").grid(row=0, column=0, padx=5)
        start_var = tk.StringVar()
        ttk.Entry(range_frame, textvariable=start_var, width=10).grid(row=0, column=1, padx=5)

        ttk.Label(range_frame, text="To:").grid(row=0, column=2, padx=5)
        end_var = tk.StringVar()
        ttk.Entry(range_frame, textvariable=end_var, width=10).grid(row=0, column=3, padx=5)

        def apply_range():
            try:
                start = float(start_var.get())
                end = float(end_var.get())

                wl = self.wavelengths
                selected = [w for w in wl if start <= w <= end]

                if not selected:
                    messagebox.showwarning("No Wavelengths",
                        f"No wavelengths found in range {start}-{end}")
                    return

                wl_text = ', '.join(map(str, selected))
                self.tab7_wl_spec.delete('1.0', 'end')
                self.tab7_wl_spec.insert('1.0', wl_text)
                self._tab7_update_wavelength_count()

                dialog.destroy()
            except ValueError:
                messagebox.showerror("Invalid Input", "Please enter valid numbers")

        ttk.Button(dialog, text="Apply", command=apply_range).pack(pady=20)

    def _tab7_reset_to_fresh(self):
        """Reset all parameters to fresh development mode."""
        self.tab7_mode_label.config(text="Status: Fresh Development", foreground='#2E7D32')

        self.tab7_config_text.config(state='normal')
        self.tab7_config_text.delete('1.0', 'end')
        self.tab7_config_text.insert('1.0', "Fresh development mode. Configure parameters below.")
        self.tab7_config_text.config(state='disabled')

        self.tab7_model_type.set('PLS')
        self.tab7_task_type.set('regression')
        self.tab7_backend.set('python')
        self.tab7_preprocess.set('raw')
        self.tab7_window.set(17)
        self.tab7_cv_folds.set(5)

        self.tab7_wl_spec.delete('1.0', 'end')
        self._tab7_update_wavelength_count()

        self._tab7_create_hyperparam_widgets('PLS')

        self.tab7_results_text.config(state='normal')
        self.tab7_results_text.delete('1.0', 'end')
        self.tab7_results_text.insert('1.0', "Run a model to see performance metrics here.")
        self.tab7_results_text.config(state='disabled')

        self.tab7_save_button.config(state='disabled')
        self.tab7_status.config(text="Reset to fresh development", foreground='#2E7D32')
