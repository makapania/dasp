def create_synthetic_data(n_samples=100, n_wavelengths=200, seed=42):
    """Create synthetic spectral data for testing."""
    np.random.seed(seed)

    wavelengths = np.linspace(1500, 1700, n_wavelengths)
    X_data = np.random.randn(n_samples, n_wavelengths) * 0.05 + 1.0

    idx1 = n_wavelengths // 4
    idx2 = n_wavelengths // 2
    idx3 = 3 * n_wavelengths // 4

    y_data = (
        2.0 * X_data[:, idx1] +
        1.5 * X_data[:, idx2] +
        1.0 * X_data[:, idx3] +
        np.random.randn(n_samples) * 0.1
    )

    X = pd.DataFrame(
        X_data,
        columns=[str(float(w)) for w in wavelengths]
    )
    y = pd.Series(y_data, name='target')

    return X, y


def simple_spxy_split(X, y, val_percentage=20.0, seed=42):
    """
    Simple validation split using SPXY-like logic.

    Returns:
        cal_indices: list of calibration indices
        val_indices: list of validation indices
    """
    np.random.seed(seed)

    n_total = len(X)
    n_val = int(n_total * val_percentage / 100.0)

    # Use simple distance-based selection
    from scipy.spatial.distance import pdist, squareform

    # Normalize X and y for distance calculation
    X_norm = (X.values - X.values.mean(axis=0)) / (X.values.std(axis=0) + 1e-10)
    y_norm = (y.values - y.mean()) / (y.std() + 1e-10)

    # Combine X and y for joint distance
    Xy = np.column_stack([X_norm, y_norm[:, np.newaxis]])

    # Compute pairwise distances
    distances = squareform(pdist(Xy, metric='euclidean'))

    # Select diverse samples (SPXY-like)
    max_dist_idx = np.unravel_index(distances.argmax(), distances.shape)
    selected = [max_dist_idx[0], max_dist_idx[1]]
    remaining = list(set(range(n_total)) - set(selected))

    while len(selected) < n_val:
        min_distances = []
        for idx in remaining:
            min_dist = min(distances[idx, s] for s in selected)
            min_distances.append((min_dist, idx))

        _, best_idx = max(min_distances)
        selected.append(best_idx)
        remaining.remove(best_idx)

    val_indices = selected
    cal_indices = [i for i in range(n_total) if i not in val_indices]

    return cal_indices, val_indices


class TestValidationSetR2Reproduction:
    """Test R² reproduction with validation sets (mimics GUI workflow)."""

    def test_basic_r2_reproduction_no_validation(self):
        """
        Baseline test: Verify R² matches without validation set.

        This should pass - establishes baseline behavior.
        """
        print("\n[TEST] Baseline: R² reproduction WITHOUT validation set")

        X, y = create_synthetic_data(n_samples=100, n_wavelengths=200)

        # RESULTS TAB: Run search on full data
        results_df = run_search(
            X, y,
            task_type='regression',
            folds=5,
            models_to_test=['PLS'],
            preprocessing_methods={'raw': True},
            enable_variable_subsets=False,
            enable_region_subsets=False
        )

        results_r2 = results_df.iloc[0]['R2']
        n_components = int(results_df.iloc[0]['LVs'])

        # MODEL DEVELOPMENT TAB: Manual CV (mimicking GUI)
        model = PLSRegression(n_components=n_components)
        cv = KFold(n_splits=5, shuffle=True, random_state=42)

        cv_scores = cross_val_score(
            model, X.values, y.values,
            cv=cv, scoring='r2'
        )
        modeldev_r2 = cv_scores.mean()

        print(f"  Results R²:     {results_r2:.4f}")
        print(f"  Model Dev R²:   {modeldev_r2:.4f}")
        print(f"  Difference:     {abs(results_r2 - modeldev_r2):.4f}")

        # Should match within tolerance
        assert abs(results_r2 - modeldev_r2) < 0.01, \
            f"Baseline R² mismatch! Results={results_r2:.4f}, ModelDev={modeldev_r2:.4f}"

        print("  ✓ PASS: Baseline R² values match")

    def test_r2_reproduction_with_validation_set(self):
        """
        CRITICAL TEST: Verify R² matches WITH validation set.

        This test mimics the exact GUI workflow:
        1. Create cal/val split using SPXY
        2. Results tab: Filter to calibration, run search
        3. Model Dev tab: Filter to calibration, manual CV
        4. Verify R² values match

        THIS TEST SHOULD FAIL if the bug exists!
        """
        print("\n[TEST] CRITICAL: R² reproduction WITH validation set (GUI workflow)")

        # Create data
        X, y = create_synthetic_data(n_samples=100, n_wavelengths=200)

        print(f"  Total samples: {len(X)}")

        # Create cal/val split (80/20)
        cal_indices, val_indices = simple_spxy_split(X, y, val_percentage=20.0)

        print(f"  Calibration samples: {len(cal_indices)}")
        print(f"  Validation samples: {len(val_indices)}")

        # === RESULTS TAB WORKFLOW ===
        # Filter to calibration samples
        X_cal_results = X.iloc[cal_indices]
        y_cal_results = y.iloc[cal_indices]

        print(f"\n  RESULTS TAB:")
        print(f"    X_cal shape: {X_cal_results.shape}")
        print(f"    X_cal index: {X_cal_results.index.tolist()[:10]}...")
        print(f"    Index is continuous: {list(X_cal_results.index) == list(range(len(X_cal_results)))}")

        # Run search
        results_df = run_search(
            X_cal_results, y_cal_results,
            task_type='regression',
            folds=5,
            models_to_test=['PLS'],
            preprocessing_methods={'raw': True},
            enable_variable_subsets=False,
            enable_region_subsets=False
        )

        results_r2 = results_df.iloc[0]['R2']
        n_components = int(results_df.iloc[0]['LVs'])

        print(f"    Results R²: {results_r2:.4f}")
        print(f"    n_components: {n_components}")

        # === MODEL DEVELOPMENT TAB WORKFLOW (WITHOUT INDEX RESET) ===
        # Filter to calibration samples
        X_cal_modeldev = X.iloc[cal_indices]
        y_cal_modeldev = y.iloc[cal_indices]

        print(f"\n  MODEL DEVELOPMENT TAB (NO INDEX RESET):")
        print(f"    X_cal shape: {X_cal_modeldev.shape}")
        print(f"    X_cal index: {X_cal_modeldev.index.tolist()[:10]}...")
        print(f"    Index is continuous: {list(X_cal_modeldev.index) == list(range(len(X_cal_modeldev)))}")

        # Manual CV (mimicking GUI without reset)
        model = PLSRegression(n_components=n_components)
        cv = KFold(n_splits=5, shuffle=True, random_state=42)

        cv_scores = cross_val_score(
            model, X_cal_modeldev.values, y_cal_modeldev.values,
            cv=cv, scoring='r2'
        )
        modeldev_r2_no_reset = cv_scores.mean()

        print(f"    Model Dev R² (no reset): {modeldev_r2_no_reset:.4f}")

        # === MODEL DEVELOPMENT TAB WORKFLOW (WITH INDEX RESET) ===
        # Filter to calibration samples AND RESET INDEX
        X_cal_modeldev_reset = X.iloc[cal_indices].reset_index(drop=True)
        y_cal_modeldev_reset = y.iloc[cal_indices].reset_index(drop=True)

        print(f"\n  MODEL DEVELOPMENT TAB (WITH INDEX RESET):")
        print(f"    X_cal shape: {X_cal_modeldev_reset.shape}")
        print(f"    X_cal index: {X_cal_modeldev_reset.index.tolist()[:10]}...")
        print(f"    Index is continuous: {list(X_cal_modeldev_reset.index) == list(range(len(X_cal_modeldev_reset)))}")

        # Manual CV (mimicking GUI with reset)
        cv_scores_reset = cross_val_score(
            model, X_cal_modeldev_reset.values, y_cal_modeldev_reset.values,
            cv=cv, scoring='r2'
        )
        modeldev_r2_with_reset = cv_scores_reset.mean()

        print(f"    Model Dev R² (with reset): {modeldev_r2_with_reset:.4f}")

        # === COMPARISON ===
        print(f"\n  COMPARISON:")
        print(f"    Results R²:                  {results_r2:.4f}")
        print(f"    Model Dev R² (no reset):     {modeldev_r2_no_reset:.4f}")
        print(f"    Model Dev R² (with reset):   {modeldev_r2_with_reset:.4f}")
        print(f"    Diff (no reset):             {abs(results_r2 - modeldev_r2_no_reset):.4f}")
        print(f"    Diff (with reset):           {abs(results_r2 - modeldev_r2_with_reset):.4f}")

        # Check if index reset fixes the issue
        if abs(results_r2 - modeldev_r2_no_reset) > 0.01:
            print(f"\n  ⚠️  BUG DETECTED: R² mismatch without index reset!")
            print(f"      This confirms the index alignment issue.")

            if abs(results_r2 - modeldev_r2_with_reset) < 0.01:
                print(f"  ✓  FIX CONFIRMED: Index reset fixes the issue!")
            else:
                print(f"  ✗  FIX FAILED: Index reset does NOT fix the issue.")
                print(f"      Need to investigate further...")
        else:
            print(f"\n  ✓  NO BUG: R² matches even without index reset")

        # The assertion - should pass with index reset
        assert abs(results_r2 - modeldev_r2_with_reset) < 0.01, \
            f"R² mismatch even with index reset! Results={results_r2:.4f}, ModelDev={modeldev_r2_with_reset:.4f}"

        print("\n  ✓ PASS: R² reproduction test passed (with index reset)")

    def test_validation_exclusion_manual_cv(self):
        """
        Test that demonstrates the INDEX ISSUE in detail.

        Shows what happens when CV is run on filtered data without index reset.
        """
        print("\n[TEST] Detailed: Index issue demonstration")

        # Simple dataset
        X, y = create_synthetic_data(n_samples=50, n_wavelengths=100)

        # Simulate validation split (select specific indices to remove)
        val_indices = [5, 15, 25, 35, 45]  # Every 10th sample
        cal_indices = [i for i in range(50) if i not in val_indices]

        print(f"  Total: {len(X)}, Cal: {len(cal_indices)}, Val: {len(val_indices)}")

        # Filter WITHOUT reset (mimics current GUI behavior)
        X_cal_no_reset = X.iloc[cal_indices]
        y_cal_no_reset = y.iloc[cal_indices]

        print(f"\n  WITHOUT index reset:")
        print(f"    DataFrame index: {X_cal_no_reset.index.tolist()}")
        print(f"    When converted to .values, indices are LOST")
        print(f"    CV will use indices 0-{len(X_cal_no_reset)-1} on the numpy array")
        print(f"    This is CORRECT for run_search() because it uses .values immediately")

        # Run CV
        model = PLSRegression(n_components=5)
        cv = KFold(n_splits=5, shuffle=True, random_state=42)

        # Show what CV indices look like
        print(f"\n    CV fold indices (on numpy array with {len(X_cal_no_reset)} samples):")
        for i, (train, test) in enumerate(cv.split(X_cal_no_reset.values, y_cal_no_reset.values)):
            print(f"      Fold {i}: train[0:5]={train[:5]}, test[0:5]={test[:5]}")

        r2_no_reset = cross_val_score(
            model, X_cal_no_reset.values, y_cal_no_reset.values,
            cv=cv, scoring='r2'
        ).mean()

        # Filter WITH reset
        X_cal_reset = X.iloc[cal_indices].reset_index(drop=True)
        y_cal_reset = y.iloc[cal_indices].reset_index(drop=True)

        print(f"\n  WITH index reset:")
        print(f"    DataFrame index: {X_cal_reset.index.tolist()}")
        print(f"    Indices are now 0-{len(X_cal_reset)-1} (continuous)")

        r2_reset = cross_val_score(
            model, X_cal_reset.values, y_cal_reset.values,
            cv=cv, scoring='r2'
        ).mean()

        print(f"\n  R² comparison:")
        print(f"    No reset: {r2_no_reset:.4f}")
        print(f"    With reset: {r2_reset:.4f}")
        print(f"    Difference: {abs(r2_no_reset - r2_reset):.4f}")

        # They should be identical because .values already removes index info
        # But if there's a bug, they might differ
        if abs(r2_no_reset - r2_reset) > 0.001:
            print(f"\n  ⚠️  UNEXPECTED: R² differs even though .values should remove index!")
            print(f"      This suggests a different issue than index alignment.")
        else:
            print(f"\n  ✓  EXPECTED: R² is identical (index doesn't matter with .values)")
