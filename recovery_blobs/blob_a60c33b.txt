### Reading Hyperparameters

```python
# Hyperparameters are stored dynamically
hyperparam_dict = {}

for param_name, var in self.tab7_hyperparam_widgets.items():
    if isinstance(var, tk.IntVar):
        hyperparam_dict[param_name] = var.get()
    elif isinstance(var, tk.StringVar):
        value = var.get()
        # Try to convert to appropriate type
        try:
            if '.' in value:
                hyperparam_dict[param_name] = float(value)
            elif value.lower() == 'none':
                hyperparam_dict[param_name] = None
            else:
                # Could be tuple like "(100,50)" for MLP
                hyperparam_dict[param_name] = ast.literal_eval(value)
        except:
            hyperparam_dict[param_name] = value

# Example for PLS:
# hyperparam_dict = {'n_components': 10}

# Example for RandomForest:
# hyperparam_dict = {'n_estimators': 100, 'max_depth': None, 'max_features': 'sqrt'}
```

## Parsing Wavelengths

### Full Wavelength Parser

```python
def parse_wavelengths(wl_text, available_wavelengths=None):
    """Parse wavelength specification text.

    Args:
        wl_text: Text from self.tab7_wl_spec
        available_wavelengths: Optional list of available wavelengths from data

    Returns:
        List of wavelengths (sorted, unique)
    """
    wavelengths = []
    lines = wl_text.strip().split('\n')

    for line in lines:
        line = line.strip()
        if not line or line == '...' or line.startswith('#'):
            continue

        # Check for range (e.g., "400-700")
        if '-' in line and not line.startswith('-'):
            parts = line.split('-')
            if len(parts) == 2:
                try:
                    start = float(parts[0].strip())
                    end = float(parts[1].strip())

                    if available_wavelengths is not None:
                        # Use only available wavelengths in range
                        wls = [wl for wl in available_wavelengths if start <= wl <= end]
                        wavelengths.extend(wls)
                    else:
                        # Generate range (assume 1nm intervals)
                        wls = np.arange(start, end + 1, 1)
                        wavelengths.extend(wls)
                except ValueError:
                    continue
        else:
            # Single wavelength
            try:
                wl = float(line)
                wavelengths.append(wl)
            except ValueError:
                continue

    # Remove duplicates and sort
    wavelengths = sorted(set(wavelengths))
    return wavelengths

# Usage
wl_text = self.tab7_wl_spec.get('1.0', 'end')
available_wls = list(self.X_original.columns) if self.X_original is not None else None
wavelengths = parse_wavelengths(wl_text, available_wls)
```

## Validating Configuration

### Pre-Run Validation

```python
def validate_tab7_config():
    """Validate configuration before running model.

    Returns:
        (bool, str): (is_valid, error_message)
    """
    # Check data loaded
    if self.X is None or self.y is None:
        return False, "No data loaded. Load data in Tab 1 first."

    # Check wavelengths
    wl_text = self.tab7_wl_spec.get('1.0', 'end').strip()
    if not wl_text:
        return False, "No wavelengths specified."

    wavelengths = parse_wavelengths(wl_text, list(self.X_original.columns))
    if len(wavelengths) == 0:
        return False, "Invalid wavelength specification."

    # Check hyperparameters
    model_type = self.tab7_model_type.get()

    if model_type in ['Ridge', 'Lasso']:
        alpha_str = self.tab7_hyperparam_widgets['alpha'].get()
        try:
            alpha = float(alpha_str)
            if alpha <= 0:
                return False, "Alpha must be positive."
        except ValueError:
            return False, "Invalid alpha value."

    elif model_type == 'MLP':
        layers_str = self.tab7_hyperparam_widgets['hidden_layer_sizes'].get()
        try:
            layers = ast.literal_eval(layers_str)
            if not isinstance(layers, tuple):
                return False, "Hidden layers must be tuple like (100,50)."
        except:
            return False, "Invalid hidden layer specification."

    # Check CV folds
    cv_folds = self.tab7_cv_folds.get()
    n_samples = len(self.y)
    if cv_folds > n_samples:
        return False, f"CV folds ({cv_folds}) exceeds number of samples ({n_samples})."

    return True, "Configuration valid."

# Usage in _tab7_run_model
is_valid, error_msg = validate_tab7_config()
if not is_valid:
    messagebox.showerror("Invalid Configuration", error_msg)
    return
```

## Loading Model from Results

### Example Load Function (Phase 3)

```python
def load_model_to_tab7(self, config_dict, rank):
    """Load model configuration into Tab 7 for refinement.

    Args:
        config_dict: Dictionary with model configuration
        rank: Rank in results table
    """
    # Switch to Tab 7
    self.notebook.select(self.tab7)

    # Update mode
    self.tab7_mode_label.config(
        text=f"Mode: Loaded from Results (Rank {rank})",
        foreground=self.colors['accent']
    )

    # Update config text
    self.tab7_config_text.config(state='normal')
    self.tab7_config_text.delete('1.0', 'end')

    config_text = f"""📊 Loaded Model Configuration

Model Type:      {config_dict['model']}
Task Type:       {config_dict.get('task', 'regression')}
Preprocessing:   {config_dict['preprocessing']}
Window Size:     {config_dict.get('window_size', 11)}
CV Folds:        {config_dict.get('cv_folds', 5)}
Wavelengths:     {len(config_dict['wavelengths'])} selected
Backend:         {config_dict.get('backend', 'python')}

Loaded from: Results Table (Rank {rank})
Original R²: {config_dict['r2']:.4f}
Original RMSE: {config_dict['rmse']:.4f}
"""
    self.tab7_config_text.insert('1.0', config_text)
    self.tab7_config_text.config(state='disabled')

    # Set parameters
    self.tab7_model_type.set(config_dict['model'])
    self.tab7_task_type.set(config_dict.get('task', 'regression'))
    self.tab7_preprocessing.set(config_dict['preprocessing'])
    self.tab7_backend.set(config_dict.get('backend', 'python'))
    self.tab7_window_size.set(config_dict.get('window_size', 11))
    self.tab7_cv_folds.set(config_dict.get('cv_folds', 5))

    # Update hyperparameters
    self._tab7_update_hyperparameters()

    # Set hyperparameter values
    if 'hyperparameters' in config_dict:
        for param, value in config_dict['hyperparameters'].items():
            if param in self.tab7_hyperparam_widgets:
                self.tab7_hyperparam_widgets[param].set(str(value))

    # Set wavelengths
    wavelengths = config_dict['wavelengths']
    wl_text = '\n'.join(map(str, wavelengths[:50]))  # First 50
    if len(wavelengths) > 50:
        wl_text += '\n...'

    self.tab7_wl_spec.delete('1.0', 'end')
    self.tab7_wl_spec.insert('1.0', wl_text)
    self._tab7_update_wavelength_count()

    # Update status
    self.tab7_status.config(
        text=f"Model loaded from Results (Rank {rank}). Modify parameters and re-run or use as-is."
    )
```

## Threading for Long Operations

### Running Model in Background Thread

```python
def _tab7_run_model(self):
    """Run model development (with threading for Phase 4)."""
    # Validate
    is_valid, error_msg = validate_tab7_config()
    if not is_valid:
        messagebox.showerror("Invalid Configuration", error_msg)
        return

    # Disable UI during run
    self.tab7_run_btn.config(state='disabled', text="⏳ Running...")
    self.tab7_status.config(text="Model training in progress...")

    # Run in background thread
    thread = threading.Thread(target=self._tab7_run_model_thread, daemon=True)
    thread.start()

def _tab7_run_model_thread(self):
    """Background thread for model execution."""
    try:
        # Get configuration
        config = self._tab7_get_config()

        # Run model (Phase 4 implementation)
        results = run_cv_model(config, self.X, self.y)

        # Update UI (must use root.after for thread safety)
        self.root.after(0, lambda: self._tab7_display_results(results))

    except Exception as e:
        error_msg = str(e)
        self.root.after(0, lambda: messagebox.showerror("Error", f"Model failed: {error_msg}"))

    finally:
        # Re-enable UI
        self.root.after(0, lambda: self.tab7_run_btn.config(state='normal', text="▶ Run Model Development"))
```
