"""
Integration helpers for variable selection methods with search.py.

Provides functions to apply different variable selection strategies within
the model search workflow.
"""

import numpy as np
from .spa import SuccessiveProjectionsAlgorithm
from .uve import UninformativeVariableElimination
from .hybrid import UVE_SPA
from .ipls import IntervalPLS


def select_variables_by_method(
    X,
    y,
    wavelengths,
    method='importance',
    n_vars=None,
    variable_counts=None,
    importances=None,
    **kwargs
):
    """
    Select variables using specified method.

    Parameters
    ----------
    X : ndarray
        Spectra data (already preprocessed if applicable)
    y : ndarray
        Target values
    wavelengths : ndarray
        Wavelength values
    method : str
        Selection method: 'importance', 'spa', 'uve', 'uve_spa', 'ipls'
    n_vars : int, optional
        Number of variables to select (for spa, uve_spa)
    variable_counts : list, optional
        List of variable counts to test (for importance method)
    importances : ndarray, optional
        Pre-computed importances (for importance method)
    **kwargs : dict
        Additional method-specific parameters

    Returns
    -------
    results : dict or list of dict
        Selection results. Structure depends on method:
        - importance: list of dicts (one per n_vars)
        - spa/uve_spa: single dict
        - uve: single dict with mask
        - ipls: dict with interval information
    """
    X = np.asarray(X)
    y = np.asarray(y).ravel()

    n_features = X.shape[1]

    if method == 'importance':
        return _select_by_importance(
            X, y, wavelengths, importances, variable_counts, n_features
        )

    elif method == 'spa':
        return _select_by_spa(X, y, wavelengths, n_vars, **kwargs)

    elif method == 'uve':
        return _select_by_uve(X, y, wavelengths, **kwargs)

    elif method == 'uve_spa':
        return _select_by_uve_spa(X, y, wavelengths, n_vars, **kwargs)

    elif method == 'ipls':
        return _select_by_ipls(X, y, wavelengths, **kwargs)

    else:
        raise ValueError(f"Unknown selection method: {method}")


def _select_by_importance(X, y, wavelengths, importances, variable_counts, n_features):
    """Select by feature importance ranking (current method)."""
    if importances is None:
        raise ValueError("Importances must be provided for importance-based selection")

    # Use user-specified variable counts, or default if not provided
    if variable_counts is None:
        user_variable_counts = [10, 20, 50, 100, 250, 500, 1000]
    else:
        user_variable_counts = variable_counts

    # Only test counts that are less than total features
    valid_variable_counts = [n for n in user_variable_counts if n < n_features]

    results = []
    for n_vars in valid_variable_counts:
        # Select top N most important features
        top_indices = np.argsort(importances)[-n_vars:][::-1]

        # Sort indices for consistent ordering
        top_indices_sorted = np.sort(top_indices)

        # Get wavelengths if available
        if wavelengths is not None and len(wavelengths) == len(importances):
            selected_wavelengths = wavelengths[top_indices_sorted]
        else:
            selected_wavelengths = None

        result = {
            'method': 'importance',
            'n_vars': n_vars,
            'selected_indices': top_indices_sorted,
            'selected_wavelengths': selected_wavelengths,
            'subset_tag': f'top{n_vars}',
            'diagnostics': {
                'importances': importances[top_indices_sorted],
                'ranking': top_indices,
            }
        }
        results.append(result)

    return results


def _select_by_spa(X, y, wavelengths, n_vars, **kwargs):
    """Select using Successive Projections Algorithm."""
    verbose = kwargs.get('verbose', False)
    n_random_starts = kwargs.get('n_random_starts', 10)

    if n_vars is None:
        raise ValueError("n_vars must be specified for SPA")

    if verbose:
        print(f"\n  === SPA Variable Selection ===")
        print(f"  Target variables: {n_vars}")

    spa = SuccessiveProjectionsAlgorithm(
        n_vars=n_vars,
        n_random_starts=n_random_starts,
        validation_method='loo',
        verbose=verbose
    )

    spa.fit(X, y)

    selected_indices = spa.selected_variables_
    selected_indices_sorted = np.sort(selected_indices)

    # Get wavelengths
    if wavelengths is not None and len(wavelengths) >= X.shape[1]:
        selected_wavelengths = wavelengths[selected_indices_sorted]
    else:
        selected_wavelengths = None

    # Calculate collinearity measure
    try:
        collinearity = spa.get_collinearity_measure(X)
    except Exception:
        collinearity = None

    result = {
        'method': 'spa',
        'n_vars': n_vars,
        'selected_indices': selected_indices_sorted,
        'selected_wavelengths': selected_wavelengths,
        'subset_tag': f'spa{n_vars}',
        'diagnostics': {
            'projection_norms': spa.projection_norms_,
            'initial_wavelength': spa.initial_wavelength_,
            'rmsecv': spa.rmsecv_,
            'collinearity_vif': collinearity,
            'spa_object': spa,  # Store for visualization
        }
    }

    return [result]  # Return as list for consistency with importance method


def _select_by_uve(X, y, wavelengths, **kwargs):
    """Select using Uninformative Variable Elimination."""
    verbose = kwargs.get('verbose', False)
    cutoff_multiplier = kwargs.get('uve_cutoff_multiplier', 1.0)
    n_components = kwargs.get('uve_n_components', None)

    if verbose:
        print(f"\n  === UVE Variable Elimination ===")
        print(f"  Cutoff multiplier: {cutoff_multiplier}")

    uve = UninformativeVariableElimination(
        cutoff_multiplier=cutoff_multiplier,
        n_components=n_components,
        verbose=verbose
    )

    uve.fit(X, y)

    selected_indices = uve.get_informative_indices()
    selected_indices_sorted = np.sort(selected_indices)

    # Get wavelengths
    if wavelengths is not None and len(wavelengths) >= X.shape[1]:
        selected_wavelengths = wavelengths[selected_indices_sorted]
    else:
        selected_wavelengths = None

    result = {
        'method': 'uve',
        'n_vars': len(selected_indices),
        'selected_indices': selected_indices_sorted,
        'selected_wavelengths': selected_wavelengths,
        'subset_tag': f'uve{len(selected_indices)}',
        'diagnostics': {
            'n_eliminated': uve.n_eliminated_,
            'n_retained': uve.n_retained_,
            'cutoff_threshold': uve.cutoff_threshold_,
            'stability_scores': uve.stability_scores_,
            'informative_mask': uve.informative_mask_,
            'uve_object': uve,  # Store for visualization
        }
    }

    return [result]  # Return as list for consistency


def _select_by_uve_spa(X, y, wavelengths, n_vars, **kwargs):
    """Select using UVE-SPA hybrid."""
    verbose = kwargs.get('verbose', False)
    cutoff_multiplier = kwargs.get('uve_cutoff_multiplier', 1.0)
    n_components = kwargs.get('uve_n_components', None)
    n_random_starts = kwargs.get('n_random_starts', 10)

    if n_vars is None:
        raise ValueError("n_vars must be specified for UVE-SPA")

    if verbose:
        print(f"\n  === UVE-SPA Hybrid Selection ===")
        print(f"  Target variables: {n_vars}")
        print(f"  UVE cutoff multiplier: {cutoff_multiplier}")

    hybrid = UVE_SPA(
        n_vars=n_vars,
        uve_cutoff_multiplier=cutoff_multiplier,
        uve_n_components=n_components,
        spa_n_random_starts=n_random_starts,
        verbose=verbose
    )

    hybrid.fit(X, y)

    selected_indices = hybrid.selected_variables_
    selected_indices_sorted = np.sort(selected_indices)

    # Get wavelengths
    if wavelengths is not None and len(wavelengths) >= X.shape[1]:
        selected_wavelengths = wavelengths[selected_indices_sorted]
    else:
        selected_wavelengths = None

    # Get summary
    summary = hybrid.get_selection_summary()

    result = {
        'method': 'uve_spa',
        'n_vars': n_vars,
        'selected_indices': selected_indices_sorted,
        'selected_wavelengths': selected_wavelengths,
        'subset_tag': f'uvespa{n_vars}',
        'diagnostics': {
            'uve_eliminated': summary['uve_eliminated'],
            'uve_retained': summary['uve_retained'],
            'spa_selected': summary['spa_selected'],
            'total_reduction_pct': summary['total_reduction_pct'],
            'hybrid_object': hybrid,  # Store for visualization
        }
    }

    return [result]  # Return as list for consistency


def _select_by_ipls(X, y, wavelengths, **kwargs):
    """Select using Interval PLS."""
    verbose = kwargs.get('verbose', False)
    n_intervals = kwargs.get('n_intervals', 20)
    max_n_components = kwargs.get('max_n_components', 10)

    if verbose:
        print(f"\n  === iPLS Interval Analysis ===")
        print(f"  Number of intervals: {n_intervals}")

    ipls = IntervalPLS(
        n_intervals=n_intervals,
        max_n_components=max_n_components,
        optimize_components=True,
        verbose=verbose
    )

    ipls.fit(X, y, wavelengths=wavelengths)

    # Get best interval
    best_interval = ipls.get_best_interval()

    selected_indices = np.arange(
        best_interval['start_idx'],
        best_interval['end_idx']
    )

    # Get wavelengths for best interval
    if wavelengths is not None and len(wavelengths) >= X.shape[1]:
        selected_wavelengths = wavelengths[selected_indices]
    else:
        selected_wavelengths = None

    # Also get top 3 intervals for alternative models
    top_intervals = ipls.get_ranked_intervals(top_n=3)

    results = []

    # Add best interval result
    result = {
        'method': 'ipls',
        'n_vars': len(selected_indices),
        'selected_indices': selected_indices,
        'selected_wavelengths': selected_wavelengths,
        'subset_tag': f'ipls_int{best_interval["interval_idx"]}',
        'diagnostics': {
            'interval_idx': best_interval['interval_idx'],
            'rmsecv': best_interval['rmsecv'],
            'optimal_n_components': best_interval['optimal_n_components'],
            'r2': best_interval['r2'],
            'ipls_object': ipls,  # Store for visualization
            'all_intervals': ipls.interval_results_,
            'full_spectrum_result': ipls.full_spectrum_result_,
        }
    }
    results.append(result)

    # Optionally add other top intervals
    for i, interval in enumerate(top_intervals[1:3], start=2):  # Skip first (already added)
        interval_indices = np.arange(interval['start_idx'], interval['end_idx'])

        if wavelengths is not None and len(wavelengths) >= X.shape[1]:
            interval_wavelengths = wavelengths[interval_indices]
        else:
            interval_wavelengths = None

        result = {
            'method': 'ipls',
            'n_vars': len(interval_indices),
            'selected_indices': interval_indices,
            'selected_wavelengths': interval_wavelengths,
            'subset_tag': f'ipls_int{interval["interval_idx"]}_rank{i}',
            'diagnostics': {
                'interval_idx': interval['interval_idx'],
                'rmsecv': interval['rmsecv'],
                'optimal_n_components': interval['optimal_n_components'],
                'r2': interval['r2'],
                'rank': i,
            }
        }
        results.append(result)

    return results


def apply_uve_prefilter(X, y, cutoff_multiplier=1.0, n_components=None, verbose=False):
    """
    Apply UVE as a pre-filter before other selection methods.

    This can be used to remove noise variables before applying importance-based
    selection or other methods.

    Parameters
    ----------
    X : ndarray
        Spectra data
    y : ndarray
        Target values
    cutoff_multiplier : float, optional
        UVE cutoff multiplier
    n_components : int, optional
        Number of PLS components for UVE
    verbose : bool, optional
        Print progress

    Returns
    -------
    X_filtered : ndarray
        Filtered data
    selected_indices : ndarray
        Indices of retained variables
    uve_obj : UninformativeVariableElimination
        Fitted UVE object
    """
    if verbose:
        print("\n  === UVE Pre-filtering ===")

    uve = UninformativeVariableElimination(
        cutoff_multiplier=cutoff_multiplier,
        n_components=n_components,
        verbose=verbose
    )

    uve.fit(X, y)

    X_filtered = uve.transform(X)
    selected_indices = uve.get_informative_indices()

    if verbose:
        print(f"  UVE filtered: {uve.n_eliminated_} eliminated, {uve.n_retained_} retained")

    return X_filtered, selected_indices, uve
