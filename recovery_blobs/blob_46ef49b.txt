### Step 2: Create UI Widgets (15 min)

Replace the placeholder `_create_tab7_model_development()` method (currently around line 1313) with:

```python
def _create_tab7_model_development(self):
    """Tab 7: Model Development - Complete implementation."""
    self.tab7 = ttk.Frame(self.notebook, style='TFrame')
    self.notebook.add(self.tab7, text='  🔬 Model Development  ')

    # Create scrollable content
    canvas = tk.Canvas(self.tab7, bg=self.colors['bg'], highlightthickness=0)
    scrollbar = ttk.Scrollbar(self.tab7, orient="vertical", command=canvas.yview)
    content_frame = ttk.Frame(canvas, style='TFrame', padding="30")

    content_frame.bind("<Configure>", lambda e: canvas.configure(scrollregion=canvas.bbox("all")))
    canvas.create_window((0, 0), window=content_frame, anchor="nw")
    canvas.configure(yscrollcommand=scrollbar.set)

    canvas.pack(side="left", fill="both", expand=True)
    scrollbar.pack(side="right", fill="y")

    # Title
    ttk.Label(content_frame, text="Model Development", style='Title.TLabel').pack(pady=20)

    # === WAVELENGTH SPECIFICATION ===
    wl_frame = ttk.LabelFrame(content_frame, text="Wavelength Selection", padding="10")
    wl_frame.pack(fill='x', pady=10)

    ttk.Label(wl_frame, text="Wavelength Specification (e.g., '1500-2500' or '1500-1800, 2000-2400'):").pack(anchor='w')
    self.tab7_wl_spec = tk.Text(wl_frame, height=3, width=60, wrap=tk.WORD)
    self.tab7_wl_spec.pack(fill='x', pady=5)
    self.tab7_wl_spec.insert('1.0', '1000-2500')  # Default full range

    # === MODEL CONFIGURATION ===
    config_frame = ttk.LabelFrame(content_frame, text="Model Configuration", padding="10")
    config_frame.pack(fill='x', pady=10)

    # Model Type
    row = 0
    ttk.Label(config_frame, text="Model Type:").grid(row=row, column=0, sticky='w', pady=5)
    self.tab7_model_type = tk.StringVar(value='PLS')
    model_combo = ttk.Combobox(config_frame, textvariable=self.tab7_model_type, width=20,
                               values=['PLS', 'Ridge', 'Lasso', 'RandomForest', 'MLP', 'NeuralBoosted'])
    model_combo.grid(row=row, column=1, sticky='w', pady=5, padx=5)

    # Task Type
    row += 1
    ttk.Label(config_frame, text="Task Type:").grid(row=row, column=0, sticky='w', pady=5)
    self.tab7_task_type = tk.StringVar(value='regression')
    task_combo = ttk.Combobox(config_frame, textvariable=self.tab7_task_type, width=20,
                              values=['regression', 'classification'])
    task_combo.grid(row=row, column=1, sticky='w', pady=5, padx=5)

    # Preprocessing
    row += 1
    ttk.Label(config_frame, text="Preprocessing:").grid(row=row, column=0, sticky='w', pady=5)
    self.tab7_preprocess = tk.StringVar(value='raw')
    preprocess_combo = ttk.Combobox(config_frame, textvariable=self.tab7_preprocess, width=20,
                                    values=['raw', 'snv', 'msc', 'sg1', 'sg2', 'snv_sg1', 'snv_sg2',
                                           'deriv_snv', 'msc_sg1', 'msc_sg2', 'deriv_msc'])
    preprocess_combo.grid(row=row, column=1, sticky='w', pady=5, padx=5)

    # Window Size
    row += 1
    ttk.Label(config_frame, text="Window Size:").grid(row=row, column=0, sticky='w', pady=5)
    self.tab7_window = tk.IntVar(value=17)
    window_spin = ttk.Spinbox(config_frame, from_=5, to=51, increment=2, textvariable=self.tab7_window, width=18)
    window_spin.grid(row=row, column=1, sticky='w', pady=5, padx=5)

    # CV Folds
    row += 1
    ttk.Label(config_frame, text="CV Folds:").grid(row=row, column=0, sticky='w', pady=5)
    self.tab7_folds = tk.IntVar(value=5)
    folds_spin = ttk.Spinbox(config_frame, from_=3, to=10, textvariable=self.tab7_folds, width=18)
    folds_spin.grid(row=row, column=1, sticky='w', pady=5, padx=5)

    # Max Iterations
    row += 1
    ttk.Label(config_frame, text="Max Iterations:").grid(row=row, column=0, sticky='w', pady=5)
    self.tab7_max_iter = tk.IntVar(value=100)
    iter_spin = ttk.Spinbox(config_frame, from_=50, to=500, increment=50, textvariable=self.tab7_max_iter, width=18)
    iter_spin.grid(row=row, column=1, sticky='w', pady=5, padx=5)

    # === ACTION BUTTONS ===
    button_frame = ttk.Frame(content_frame)
    button_frame.pack(pady=20)

    self.tab7_run_button = ttk.Button(button_frame, text='▶ Run Model',
                                       command=self._tab7_run_model,
                                       style='Accent.TButton')
    self.tab7_run_button.pack(side='left', padx=5)

    self.tab7_save_button = ttk.Button(button_frame, text='💾 Save Model',
                                        command=self._tab7_save_model,
                                        state='disabled')
    self.tab7_save_button.pack(side='left', padx=5)

    # === STATUS ===
    self.tab7_status = ttk.Label(content_frame, text='Ready to develop model', style='Caption.TLabel')
    self.tab7_status.pack(pady=10)

    # === RESULTS ===
    results_frame = ttk.LabelFrame(content_frame, text="Results", padding="10")
    results_frame.pack(fill='both', expand=True, pady=10)

    self.tab7_results_text = tk.Text(results_frame, height=20, width=80, wrap=tk.WORD,
                                      state='disabled', font=('Consolas', 10))
    results_scrollbar = ttk.Scrollbar(results_frame, command=self.tab7_results_text.yview)
    self.tab7_results_text.configure(yscrollcommand=results_scrollbar.set)

    self.tab7_results_text.pack(side='left', fill='both', expand=True)
    results_scrollbar.pack(side='right', fill='y')
```

## Common Integration Issues

### Issue 1: Import Errors

**Symptom:** `ImportError: cannot import name 'get_model'`

**Fix:** Ensure imports at top of file:
```python
from spectral_predict.models import get_model
from spectral_predict.preprocess import build_preprocessing_pipeline
```

---

### Issue 2: Attribute Errors

**Symptom:** `AttributeError: 'SpectralPredictApp' object has no attribute 'tab7_wl_spec'`

**Fix:** Ensure Step 2 (Create UI Widgets) is completed and widgets are created before methods are called.

---

### Issue 3: Method Not Found

**Symptom:** `AttributeError: 'SpectralPredictApp' object has no attribute '_tab7_run_model'`

**Fix:** Ensure Step 1 (Copy 5 Methods) is completed and methods are added to the class (check indentation).
