"""
Hybrid variable selection methods combining multiple approaches.

Based on:
    Ye et al. (2008). "A combination of chemometric methods and GC-MS for
    the classification of edible vegetable oils."
    Chemometrics and Intelligent Laboratory Systems, 91: 194-199.

    Xiaobo et al. (2010). Section 4.3.3, page 23.

The UVE-SPA hybrid combines:
    1. UVE (Uninformative Variable Elimination) - removes noisy variables
    2. SPA (Successive Projections Algorithm) - selects non-collinear subset

This two-stage approach addresses both noise AND collinearity.
"""

import numpy as np
from sklearn.base import BaseEstimator
from .uve import UninformativeVariableElimination
from .spa import SuccessiveProjectionsAlgorithm


class UVE_SPA(BaseEstimator):
    """
    Hybrid UVE-SPA variable selection.

    Two-stage process:
        Stage 1: Apply UVE to filter out uninformative (noisy) variables
        Stage 2: Apply SPA to select non-collinear subset from filtered variables

    This addresses both major issues:
        - UVE handles noise
        - SPA handles collinearity

    Parameters
    ----------
    n_vars : int
        Final number of variables to select (passed to SPA)

    uve_cutoff_multiplier : float, optional (default=1.0)
        Cutoff multiplier for UVE stage. Range 0.7-1.5 typical.
        Higher = more aggressive filtering.

    uve_n_components : int, optional (default=None)
        Number of PLS components for UVE. If None, determined automatically.

    spa_n_random_starts : int, optional (default=10)
        Number of random starting wavelengths for SPA

    verbose : bool, optional (default=False)
        Whether to print progress information

    Attributes
    ----------
    uve_ : UninformativeVariableElimination
        Fitted UVE object from stage 1

    spa_ : SuccessiveProjectionsAlgorithm
        Fitted SPA object from stage 2

    selected_variables_ : ndarray
        Final selected variable indices (in original feature space)

    uve_filtered_indices_ : ndarray
        Indices retained after UVE filtering

    spa_selected_indices_ : ndarray
        Indices selected by SPA (in UVE-filtered space)

    n_eliminated_by_uve_ : int
        Number of variables eliminated by UVE

    n_selected_by_spa_ : int
        Number of variables selected by SPA
    """

    def __init__(self, n_vars, uve_cutoff_multiplier=1.0,
                 uve_n_components=None, spa_n_random_starts=10,
                 verbose=False):
        self.n_vars = n_vars
        self.uve_cutoff_multiplier = uve_cutoff_multiplier
        self.uve_n_components = uve_n_components
        self.spa_n_random_starts = spa_n_random_starts
        self.verbose = verbose

    def fit(self, X, y):
        """
        Apply UVE-SPA hybrid selection.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            Spectra data
        y : ndarray of shape (n_samples,)
            Target values

        Returns
        -------
        self : object
            Fitted estimator
        """
        X = np.asarray(X)
        y = np.asarray(y).ravel()

        n_samples, n_features = X.shape

        if self.verbose:
            print("=" * 60)
            print("UVE-SPA HYBRID VARIABLE SELECTION")
            print("=" * 60)
            print(f"Initial variables: {n_features}")
            print(f"Target final variables: {self.n_vars}")

        # Validation
        if self.n_vars < 1:
            raise ValueError(f"n_vars must be >= 1, got {self.n_vars}")
        if self.n_vars >= n_features:
            raise ValueError(
                f"n_vars ({self.n_vars}) must be < n_features ({n_features})"
            )

        # ===== STAGE 1: UVE FILTERING =====
        if self.verbose:
            print("\n" + "-" * 60)
            print("STAGE 1: UVE (Uninformative Variable Elimination)")
            print("-" * 60)

        self.uve_ = UninformativeVariableElimination(
            cutoff_multiplier=self.uve_cutoff_multiplier,
            n_components=self.uve_n_components,
            verbose=self.verbose
        )

        self.uve_.fit(X, y)

        # Get filtered variables
        X_filtered = self.uve_.transform(X)
        uve_filtered_indices = self.uve_.get_informative_indices()

        self.uve_filtered_indices_ = uve_filtered_indices
        self.n_eliminated_by_uve_ = n_features - len(uve_filtered_indices)

        if self.verbose:
            print(f"\nUVE Stage Results:")
            print(f"  Eliminated: {self.n_eliminated_by_uve_} variables")
            print(f"  Retained: {len(uve_filtered_indices)} variables")

        # Check if UVE left enough variables for SPA
        if len(uve_filtered_indices) <= self.n_vars:
            if self.verbose:
                print(f"\nWARNING: UVE retained only {len(uve_filtered_indices)} "
                      f"variables, but n_vars={self.n_vars}.")
                print("Skipping SPA stage - using all UVE-filtered variables.")

            self.spa_ = None
            self.selected_variables_ = uve_filtered_indices
            self.spa_selected_indices_ = np.arange(len(uve_filtered_indices))
            self.n_selected_by_spa_ = len(uve_filtered_indices)

            return self

        # ===== STAGE 2: SPA SELECTION =====
        if self.verbose:
            print("\n" + "-" * 60)
            print("STAGE 2: SPA (Successive Projections Algorithm)")
            print("-" * 60)
            print(f"Input: {X_filtered.shape[1]} UVE-filtered variables")
            print(f"Selecting: {self.n_vars} non-collinear variables")

        self.spa_ = SuccessiveProjectionsAlgorithm(
            n_vars=self.n_vars,
            n_random_starts=self.spa_n_random_starts,
            validation_method='loo',
            verbose=self.verbose
        )

        self.spa_.fit(X_filtered, y)

        # Get SPA selected indices (in filtered space)
        spa_selected_in_filtered_space = self.spa_.selected_variables_

        # Map back to original feature space
        selected_in_original_space = uve_filtered_indices[spa_selected_in_filtered_space]

        # Store results
        self.spa_selected_indices_ = spa_selected_in_filtered_space
        self.selected_variables_ = selected_in_original_space
        self.n_selected_by_spa_ = len(selected_in_original_space)

        if self.verbose:
            print(f"\nSPA Stage Results:")
            print(f"  Selected: {self.n_selected_by_spa_} variables")
            print("\n" + "=" * 60)
            print("HYBRID SELECTION COMPLETE")
            print("=" * 60)
            print(f"Overall reduction: {n_features} → {self.n_selected_by_spa_}")
            print(f"Reduction rate: {100*(1 - self.n_selected_by_spa_/n_features):.1f}%")

        return self

    def transform(self, X):
        """
        Select the final chosen variables from X.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            Spectra data

        Returns
        -------
        X_selected : ndarray of shape (n_samples, n_vars)
            Selected variables
        """
        if not hasattr(self, 'selected_variables_'):
            raise RuntimeError("Must call fit() before transform()")

        return X[:, self.selected_variables_]

    def fit_transform(self, X, y):
        """
        Fit and transform in one step.

        Parameters
        ----------
        X : ndarray
            Spectra data
        y : ndarray
            Target values

        Returns
        -------
        X_selected : ndarray
            Selected variables
        """
        self.fit(X, y)
        return self.transform(X)

    def get_selection_summary(self):
        """
        Get summary of the selection process.

        Returns
        -------
        summary : dict
            Dictionary with selection details:
            - 'initial_n_features': Original number of features
            - 'uve_eliminated': Variables eliminated by UVE
            - 'uve_retained': Variables retained by UVE
            - 'spa_selected': Variables selected by SPA
            - 'final_n_features': Final number of features
            - 'total_reduction_pct': Overall reduction percentage
            - 'uve_stage_reduction_pct': UVE stage reduction
            - 'spa_stage_reduction_pct': SPA stage reduction
        """
        if not hasattr(self, 'selected_variables_'):
            raise RuntimeError("Must call fit() before get_selection_summary()")

        initial_n = len(self.uve_filtered_indices_) + self.n_eliminated_by_uve_

        summary = {
            'initial_n_features': initial_n,
            'uve_eliminated': self.n_eliminated_by_uve_,
            'uve_retained': len(self.uve_filtered_indices_),
            'spa_selected': self.n_selected_by_spa_,
            'final_n_features': len(self.selected_variables_),
            'total_reduction_pct': 100 * (1 - len(self.selected_variables_) / initial_n),
            'uve_stage_reduction_pct': 100 * (self.n_eliminated_by_uve_ / initial_n),
            'spa_stage_reduction_pct': 100 * (1 - self.n_selected_by_spa_ / len(self.uve_filtered_indices_)),
        }

        return summary

    def get_diagnostics(self):
        """
        Get detailed diagnostics from both stages.

        Returns
        -------
        diagnostics : dict
            Dictionary containing:
            - 'uve_diagnostics': Diagnostics from UVE stage
            - 'spa_collinearity': Collinearity measure from SPA
            - 'selection_summary': Overall summary
        """
        if not hasattr(self, 'selected_variables_'):
            raise RuntimeError("Must call fit() before get_diagnostics()")

        diagnostics = {
            'uve_diagnostics': self.uve_.get_stability_diagnostics(),
            'selection_summary': self.get_selection_summary(),
        }

        # Add SPA diagnostics if SPA was used
        if self.spa_ is not None:
            diagnostics['spa_projection_norms'] = self.spa_.projection_norms_
            diagnostics['spa_initial_wavelength'] = self.spa_.initial_wavelength_
            diagnostics['spa_rmsecv'] = self.spa_.rmsecv_

        return diagnostics
