## CRITICAL BUG #2: Julia NeuralBoosted - All 200 Learners Fail

### Root Cause
**Julia uses Adam optimizer, Python uses LBFGS optimizer**

| Aspect | Python (Working) | Julia (Broken) |
|--------|------------------|----------------|
| **Optimizer** | LBFGS (L-BFGS quasi-Newton) | Adam (gradient descent) |
| **Learning Rate** | Adaptive (line search) | Fixed 0.01 |
| **Convergence** | 10-30 iterations | Needs 100-500 iterations |
| **Tolerance** | 5e-4 (relaxed) | 1e-4 (strict) |
| **Design For** | Small networks (3-5 neurons) | Deep networks (100+ neurons) |

### Python Implementation (WORKING)
**File:** `src/spectral_predict/neural_boosted.py`
**Key Details:**
- Line 236: `solver='lbfgs'` - Uses sklearn's LBFGS
- Line 236: `tol=5e-4` - Relaxed tolerance
- Line 234: `random_state=self.random_state + i` - Unique seed per learner
- Line 236: `max_iter=100` - Usually converges in 10-30

### Julia Implementation (BROKEN)
**File:** `julia_port/SpectralPredict/src/neural_boosted.jl`
**Problems:**
- Line 279: `opt = Adam(0.01)` - Wrong optimizer
- Line 282-328: Training loop uses Adam
- Line 317: `tolerance = 1e-4` - Too strict
- Requires reimplementation with LBFGS from Optim.jl

### Step 2: Rewrite Training Function
**File:** `julia_port/SpectralPredict/src/neural_boosted.jl`

**Replace lines 273-328** (current Adam-based training) with LBFGS implementation:

```julia
using Optim

function train_weak_learner!(
    model::Chain,
    X::Matrix{Float64},
    y::Vector{Float64},
    max_iter::Int=100,
    alpha::Float64=0.0001,
    verbose::Int=0
)::Bool
    """
    Train weak learner using LBFGS optimizer (matches sklearn).

    Args:
        model: Flux Chain to train
        X: Training data (n_features × n_samples)
        y: Target values (n_samples,)
        max_iter: Maximum LBFGS iterations (default: 100)
        alpha: L2 regularization parameter (default: 0.0001)
        verbose: Verbosity level (0=silent, 1=warnings, 2=debug)

    Returns:
        success: true if training succeeded, false otherwise
    """

    try
        # Prepare data (Flux expects features × samples)
        X_t = Float64.(X')
        y_t = Float64.(reshape(y, 1, :))

        # Flatten model parameters for LBFGS
        ps, re = Flux.destructure(model)

        # Define loss function for LBFGS
        function loss_fn(params::Vector{Float64})
            # Reconstruct model from flattened parameters
            m = re(params)

            # Forward pass
            pred = m(X_t)

            # MSE loss + L2 regularization
            mse = Flux.mse(pred, y_t)
            l2_penalty = sum(params .^ 2)

            total_loss = mse + alpha * l2_penalty

            # Check for NaN/Inf
            if !isfinite(total_loss)
                return Inf
            end

            return total_loss
        end

        # Define gradient function
        function gradient_fn!(G::Vector{Float64}, params::Vector{Float64})
            # Compute gradient using automatic differentiation
            grad = Zygote.gradient(loss_fn, params)[1]

            if grad === nothing || any(!isfinite, grad)
                fill!(G, 0.0)
                return
            end

            G .= grad
        end

        # Run LBFGS optimization
        result = optimize(
            loss_fn,
            gradient_fn!,
            ps,
            LBFGS(),
            Optim.Options(
                iterations=max_iter,
                f_tol=5e-4,  # Match sklearn tolerance
                g_tol=1e-5,
                show_trace=(verbose >= 2),
                store_trace=false,
                extended_trace=false
            )
        )

        # Check convergence
        if !Optim.converged(result)
            if verbose >= 1
                @warn "LBFGS did not converge" iterations=Optim.iterations(result) f_minimum=Optim.minimum(result)
            end
            # Continue anyway - partial convergence may be acceptable
        end

        # Update model with optimized parameters
        optimal_params = Optim.minimizer(result)
        model_optimized = re(optimal_params)

        # Copy optimized parameters back to original model
        Flux.loadparams!(model, Flux.params(model_optimized))

        # Validate predictions
        pred = model(X_t)
        if any(!isfinite, pred)
            if verbose >= 1
                @warn "Model produces non-finite predictions after training"
            end
            return false
        end

        if verbose >= 2
            final_loss = loss_fn(optimal_params)
            println("  ✓ LBFGS converged: $(Optim.iterations(result)) iterations, loss=$(final_loss)")
        end

        return true

    catch e
        if verbose >= 1
            @warn "Training failed with exception" exception=e
        end
        return false
    end
end
```
