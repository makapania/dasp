"""
Uninformative Variable Elimination (UVE) for noise filtering.

Based on:
    Centner et al. (1996). "Elimination of uninformative variables for
    multivariate calibration." Analytical Chemistry, 68: 3851-3858.

    Xiaobo et al. (2010). Section 4.3.2, pages 22-23.

UVE filters out variables where noise dominates the signal by comparing
the stability of regression coefficients to artificial random variables.
This provides an objective threshold for noise elimination.
"""

import numpy as np
from sklearn.base import BaseEstimator
from sklearn.cross_decomposition import PLSRegression
from sklearn.model_selection import LeaveOneOut
import warnings


class UninformativeVariableElimination(BaseEstimator):
    """
    Uninformative Variable Elimination for noise-based variable filtering.

    UVE adds artificial noise variables as a reference, then eliminates
    variables whose stability (mean/std of regression coefficients) is
    lower than the maximum noise variable stability.

    Parameters
    ----------
    cutoff_multiplier : float, optional (default=1.0)
        Multiplier for cutoff threshold. Default 1.0 uses max(|s_noise|).
        Values 0.7-1.5 are typical. Higher values = more aggressive filtering.

    n_components : int, optional (default=None)
        Number of PLS components. If None, determined automatically via
        cross-validation (up to max_n_components).

    max_n_components : int, optional (default=10)
        Maximum number of PLS components to consider if n_components=None.

    noise_amplitude : float, optional (default=1e-10)
        Amplitude of artificial noise variables. Should be small to not
        influence model but large enough to be numerically stable.

    random_state : int, optional (default=42)
        Random seed for noise generation (reproducibility).

    verbose : bool, optional (default=False)
        Whether to print progress information.

    Attributes
    ----------
    informative_mask_ : ndarray of shape (n_features,)
        Boolean mask: True for informative variables, False for uninformative

    stability_scores_ : ndarray of shape (n_features,)
        Stability score (mean/std) for each variable

    cutoff_threshold_ : float
        Threshold used for elimination

    noise_stability_scores_ : ndarray of shape (n_features,)
        Stability scores of artificial noise variables

    n_components_ : int
        Number of PLS components used

    n_eliminated_ : int
        Number of variables eliminated

    n_retained_ : int
        Number of variables retained
    """

    def __init__(self, cutoff_multiplier=1.0, n_components=None,
                 max_n_components=10, noise_amplitude=1e-10,
                 random_state=42, verbose=False):
        self.cutoff_multiplier = cutoff_multiplier
        self.n_components = n_components
        self.max_n_components = max_n_components
        self.noise_amplitude = noise_amplitude
        self.random_state = random_state
        self.verbose = verbose

    def _determine_optimal_components(self, X, y):
        """
        Determine optimal number of PLS components via cross-validation.

        Parameters
        ----------
        X : ndarray
            Spectra data
        y : ndarray
            Target values

        Returns
        -------
        n_opt : int
            Optimal number of components
        """
        from sklearn.metrics import mean_squared_error
        from sklearn.model_selection import cross_val_predict

        n_samples = X.shape[0]
        max_comp = min(self.max_n_components, n_samples - 1, X.shape[1])

        best_rmse = np.inf
        best_n = 1

        for n in range(1, max_comp + 1):
            try:
                pls = PLSRegression(n_components=n)
                y_pred = cross_val_predict(pls, X, y, cv=min(5, n_samples))
                rmse = np.sqrt(mean_squared_error(y, y_pred))

                if rmse < best_rmse:
                    best_rmse = rmse
                    best_n = n

            except Exception:
                break

        if self.verbose:
            print(f"UVE: Optimal n_components = {best_n} (RMSECV = {best_rmse:.6f})")

        return best_n

    def fit(self, X, y):
        """
        Identify uninformative variables using UVE.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            Spectra data
        y : ndarray of shape (n_samples,)
            Target values

        Returns
        -------
        self : object
            Fitted estimator
        """
        X = np.asarray(X)
        y = np.asarray(y).ravel()

        n_samples, n_features = X.shape

        # Validation
        if n_samples < 3:
            raise ValueError(f"Need at least 3 samples, got {n_samples}")

        # Determine number of components
        if self.n_components is None:
            n_comp = self._determine_optimal_components(X, y)
        else:
            n_comp = min(self.n_components, n_samples - 1, n_features)

        self.n_components_ = n_comp

        if self.verbose:
            print(f"UVE: Using {n_comp} PLS components")

        # Generate artificial noise matrix
        np.random.seed(self.random_state)
        noise_matrix = np.random.randn(n_samples, n_features) * self.noise_amplitude

        # Create extended matrix [X | Noise]
        X_extended = np.hstack([X, noise_matrix])

        if self.verbose:
            print(f"UVE: Extended matrix shape: {X_extended.shape}")
            print(f"UVE: Original features: {n_features}, Noise features: {n_features}")

        # Leave-one-out cross-validation to get coefficient matrix
        loo = LeaveOneOut()
        n_splits = loo.get_n_splits(X_extended)

        # Store coefficients for each LOO iteration
        coefficients = np.zeros((n_splits, X_extended.shape[1]))

        if self.verbose:
            print(f"UVE: Running LOO cross-validation ({n_splits} iterations)...")

        for i, (train_idx, test_idx) in enumerate(loo.split(X_extended)):
            X_train = X_extended[train_idx]
            y_train = y[train_idx]

            try:
                # Build PLS model
                pls = PLSRegression(n_components=n_comp)
                pls.fit(X_train, y_train)

                # Store coefficients
                coefficients[i, :] = pls.coef_.ravel()

            except Exception as e:
                warnings.warn(f"LOO iteration {i} failed: {e}")
                coefficients[i, :] = 0.0

        # Calculate stability scores: sj = mean(bj) / std(bj)
        with warnings.catch_warnings():
            warnings.filterwarnings('ignore', category=RuntimeWarning)

            means = np.mean(coefficients, axis=0)
            stds = np.std(coefficients, axis=0)

            # Avoid division by zero
            stds[stds == 0] = np.inf

            stability = means / stds

        # Split stability scores: original vs noise
        original_stability = stability[:n_features]
        noise_stability = stability[n_features:]

        # Calculate cutoff threshold
        max_noise_stability = np.max(np.abs(noise_stability))
        cutoff = self.cutoff_multiplier * max_noise_stability

        if self.verbose:
            print(f"UVE: Max noise stability = {max_noise_stability:.6f}")
            print(f"UVE: Cutoff threshold = {cutoff:.6f}")

        # Create mask: variables with |stability| > cutoff are informative
        informative_mask = np.abs(original_stability) > cutoff

        # Store results
        self.informative_mask_ = informative_mask
        self.stability_scores_ = original_stability
        self.cutoff_threshold_ = cutoff
        self.noise_stability_scores_ = noise_stability
        self.n_eliminated_ = int(np.sum(~informative_mask))
        self.n_retained_ = int(np.sum(informative_mask))

        if self.verbose:
            print(f"UVE: Variables eliminated: {self.n_eliminated_}")
            print(f"UVE: Variables retained: {self.n_retained_}")
            print(f"UVE: Retention rate: {100*self.n_retained_/n_features:.1f}%")

        if self.n_retained_ == 0:
            warnings.warn(
                "UVE eliminated ALL variables! Try lower cutoff_multiplier. "
                "Keeping all variables."
            )
            self.informative_mask_ = np.ones(n_features, dtype=bool)
            self.n_retained_ = n_features
            self.n_eliminated_ = 0

        return self

    def transform(self, X):
        """
        Keep only informative variables.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            Spectra data

        Returns
        -------
        X_filtered : ndarray of shape (n_samples, n_retained)
            Filtered data with uninformative variables removed
        """
        if not hasattr(self, 'informative_mask_'):
            raise RuntimeError("Must call fit() before transform()")

        return X[:, self.informative_mask_]

    def fit_transform(self, X, y):
        """
        Fit and transform in one step.

        Parameters
        ----------
        X : ndarray
            Spectra data
        y : ndarray
            Target values

        Returns
        -------
        X_filtered : ndarray
            Filtered data
        """
        self.fit(X, y)
        return self.transform(X)

    def get_informative_indices(self):
        """
        Get indices of informative variables.

        Returns
        -------
        indices : ndarray
            Indices of variables that passed UVE filtering
        """
        if not hasattr(self, 'informative_mask_'):
            raise RuntimeError("Must call fit() before get_informative_indices()")

        return np.where(self.informative_mask_)[0]

    def get_stability_diagnostics(self):
        """
        Get diagnostic information about stability scores.

        Returns
        -------
        diagnostics : dict
            Dictionary containing:
            - 'original_stability': Stability scores for original variables
            - 'noise_stability': Stability scores for noise variables
            - 'cutoff': Cutoff threshold
            - 'informative_mask': Boolean mask of informative variables
            - 'n_eliminated': Number eliminated
            - 'n_retained': Number retained
        """
        if not hasattr(self, 'informative_mask_'):
            raise RuntimeError("Must call fit() before get_stability_diagnostics()")

        return {
            'original_stability': self.stability_scores_,
            'noise_stability': self.noise_stability_scores_,
            'cutoff': self.cutoff_threshold_,
            'informative_mask': self.informative_mask_,
            'n_eliminated': self.n_eliminated_,
            'n_retained': self.n_retained_,
        }
